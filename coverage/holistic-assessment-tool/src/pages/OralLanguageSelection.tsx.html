import React from 'react';
import { useNavigate } from 'react-router-dom';
import { motion } from 'framer-motion';
import { ROUTES, categoryRoute } from '../routes';
import FloatingBackground from '../components/FloatingBackground';
 
interface OralLanguageOption {
  id: string;
  title: string;
  subtitle: string;
  description: string;
  icon: string;
  route: string;
  color: string;
}
 
const OralLanguageSelection: React.FC = () => {
  const navigate = useNavigate();
 
  const oralLanguageOptions: OralLanguageOption[] = [
    {
      id: 'sentence-building',
      title: 'Sentence Building (Picture Talk)',
      subtitle: 'Sentence Formation',
      description: 'Understand sentence structure and meaning from pictures',
      icon: '💬',
      route: ROUTES.ORAL_SENTENCE_BUILDING,
      color: 'bg-green-500'
    },
    {
      id: 'story-sequencing',
      title: 'Story Sequencing (Put in Order)',
      subtitle: 'Narrative Sequencing',
      description: 'Arrange pictures in the correct story sequence',
      icon: '📖',
      route: ROUTES.ORAL_STORY_SEQUENCING,
      color: 'bg-purple-500'
    },
    {
      id: 'story-retell',
      title: 'Story Retell',
      subtitle: 'Narrative Skills',
      description: 'Read a story and retell it using Beginning, Middle, and End',
      icon: '📚',
      route: ROUTES.ORAL_STORY_RETELL,
      color: 'bg-indigo-500'
    }
  ];
 
  const handleOptionClick = (option: OralLanguageOption) => {
    // Each quiz now has its own unique route
    navigate(option.route, { state: { categoryId: option.id } });
  };
 
  return (
    <div className="min-h-screen bg-blue-50 relative overflow-hidden py-12">
      <FloatingBackground />
      <div className="container mx-auto px-4 max-w-6xl relative z-10">
        {/* Header */}
        <motion.div
          initial={{ opacity: 0, y: -20 }}
          animate={{ opacity: 1, y: 0 }}
          className="text-center mb-12"
        >
          <h1 className="text-4xl md:text-5xl font-bold text-gray-800 mb-4">
            Oral Language Assessments
          </h1>
          <p className="text-lg text-gray-600 max-w-2xl mx-auto">
            Choose an assessment to evaluate oral language comprehension and expression skills
          </p>
        </motion.div>
 
        {/* Assessment Options Grid */}
        <div className="grid md:grid-cols-2 gap-6 max-w-5xl mx-auto">
          {oralLanguageOptions.map((option, index) => (
            <motion.div
              key={option.id}
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: index * 0.1 }}
              whileHover={{ scale: 1.02 }}
              whileTap={{ scale: 0.98 }}
              onClick={() => handleOptionClick(option)}
              className="bg-white rounded-2xl shadow-xl p-8 cursor-pointer hover:shadow-2xl transition-all duration-300 border-2 border-gray-100"
            >
              {/* Icon & Title */}
              <div className="text-center mb-6">
                <div className={`w-20 h-20 ${option.color} rounded-full flex items-center justify-center mx-auto mb-4 shadow-lg`}>
                  <span className="text-4xl">{option.icon}</span>
                </div>
                <h3 className="text-2xl font-bold text-gray-800 mb-1">
                  {option.title}
                </h3>
                <p className="text-xs text-gray-500 font-medium bg-gray-100 inline-block px-2 py-0.5 rounded mb-2">
                  ({option.subtitle})
                </p>
                <p className="text-gray-600 text-sm">
                  {option.description}
                </p>
              </div>
 
              {/* Start Button */}
              <motion.button
                whileHover={{ scale: 1.05 }}
                whileTap={{ scale: 0.95 }}
                className={`w-full ${option.color} hover:brightness-110 text-white py-3 rounded-xl font-semibold shadow-lg hover:shadow-xl transition-all duration-300`}
              >
                Start Assessment
              </motion.button>
            </motion.div>
          ))}
        </div>
 
        {/* Back Button */}
        <motion.div
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          transition={{ delay: 0.3 }}
          className="text-center mt-12"
        >
          <button
            onClick={() => navigate(categoryRoute('oral-skills'))}
            className="px-8 py-3 bg-gray-200 hover:bg-gray-300 text-gray-700 font-semibold rounded-xl transition-colors duration-300 shadow-md hover:shadow-lg"
          >
            ← Back to Categories
          </button>
        </motion.div>
      </div>
    </div>
  );
};
 
export default OralLanguageSelection;
